##Usando a técnica KNN Imputer para preencher dados ausentes
#Isolando as colunas numéricas sem identificadores
cols = dados2025.select_dtypes(include=['number']).columns.drop('Matrícula', errors='ignore')

#Instancia o imputador baseado nos 5 vizinhos mais próximos
imputer = KNNImputer(n_neighbors=5)

#Aplicando a imputação
dados2025[cols] = imputer.fit_transform(dados2025[cols])

##Montando um Dataframe
# Selecionando apenas as colunas com as notas médias e faltas totais no ano
# X2025 será o conjunto de dados utilizado para o agrupamento
X2025_6 = dados2025_etapa_6[['Transporte_2025_1', 'BF_2025_1', 'M_Leitura_2025', 'M_Registro_2025', 'M_Gênero_2025', 'M_Coecoesão_2025', 'M_Português_2025', 'F_Português_2025', 'M_Matemática_2025', 'F_Matemática_2025', 'M_Ciências_2025', 'F_Ciências_2025', 'M_História_2025', 'F_História_2025', 'M_Geografia_2025', 'F_Geografia_2025', 'M_Arte_2025', 'F_Arte_2025', 'M_EdFísica_2025', 'F_EdFísica_2025', 'M_Inglês_2025', 'F_Inglês_2025' ]]
#Ideia do nome: dados usados para o agrupamento em 2025 para o 6º ano

##Padronizando as notas
# Padronizando para o agrupamento
scaler = StandardScaler()
X2025_6_scaled = scaler.fit_transform(X2025_6)

##Gráfico de cotovelo
# Encontrando o número de grupos, conforme Elbow Method (cotovelo)
wcss = []#lista com as distâncias ao quadrado de cada ponto ao centro do grupo
for i in range(1, 11):
    kmeans = KMeans(n_clusters=i, init='k-means++', random_state=42)#variar o número
    kmeans.fit(X2025_6_scaled)
    wcss.append(kmeans.inertia_)

plt.plot(range(1, 11), wcss)
plt.title('Método do Cotovelo')
plt.xlabel('Número de Clusters')
plt.ylabel('WCSS (Inércia)')
plt.show()

##Tabela com as métricas para auxiliar na escolha da quantidade de agrupamentos
metricas = []

for k in range(2, 6):
    km = KMeans(
        n_clusters=k,
        init='k-means++',
        n_init=25,
        max_iter=500,
        tol=0.00005,
        random_state=42,
    )
    labels = km.fit_predict(X2025_6_scaled)

    metricas.append({
        'k': k,
        'Silhouette (↑)': round(silhouette_score(X2025_6_scaled, labels), 3),
        'Davies-Bouldin (↓)': round(
            davies_bouldin_score(X2025_6_scaled, labels), 3
        ),
        'Calinski-Harabasz (↑)': round(
            calinski_harabasz_score(X2025_6_scaled, labels), 1
        ),
    })

tabela_validacao = pd.DataFrame(metricas)
display(tabela_validacao)

##Montando 3 grupos
# Aplicando o K-Média (K-Means) para 3 grupos
kmeans = KMeans(
    n_clusters=3,
    init='k-means++',
    n_init = 25, #quantidade de vezes que a montagem ocorrerá e que será escolhida a melhor
    max_iter = 500, #número máximo de iterações em cada execução para os centroides convergirem.
    tol = 0.00005, #critério de parada
    random_state=42)#mudar este número e o número de grupos

# Criando uma cópia do DataFrame correto (dados2025_etapa_6) para adicionar a coluna 'Grupo6_2025'
dados6_2025 = dados2025_etapa_6.copy()
#Ideia do nome: dados do 6º ano de 2025
dados6_2025['Grupo6_2025'] = kmeans.fit_predict(X2025_6_scaled)
#Ideia do nome da nova coluna: Agrupamentos realizados no 6º ano de 2025

##Técnicas
